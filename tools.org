  ;; === Java File & Class Creation (IntelliJ-style) ===

  (defun my/java-source-root ()
    "Find the Java source root (src/main/java or src/test/java)."
    (let* ((file (or (buffer-file-name) default-directory))
           (root (project-root (project-current t))))
      (catch 'found
        (dolist (dir '("src/main/java/" "src/test/java/" "src/"))
          (let ((full (expand-file-name dir root)))
            (when (and (file-directory-p full)
                       (string-prefix-p full file))
              (throw 'found full))))
        (expand-file-name "src/main/java/" root))))

  (defun my/java-current-package ()
    "Infer package from current file or directory relative to source root."
    (let* ((src-root (my/java-source-root))
           (current (or (and (buffer-file-name) (file-name-directory (buffer-file-name)))
                        default-directory))
           (relative (file-relative-name current src-root)))
      (if (or (string= relative "./") (string= relative "."))
          ""
        (replace-regexp-in-string "/" "." (directory-file-name relative)))))

  (defun my/java-create-class (fqcn kind)
    "Create a Java file for FQCN (e.g. com.foo.MyClass) of KIND."
    (let* ((parts (split-string fqcn "\\."))
           (class-name (car (last parts)))
           (package (mapconcat #'identity (butlast parts) "."))
           (src-root (my/java-source-root))
           (dir (expand-file-name (replace-regexp-in-string "\\." "/" (if (string-empty-p package) "" package)) src-root))
           (file (expand-file-name (concat class-name ".java") dir))
           (body (pcase kind
                   ('class      (format "public class %s {\n\n}" class-name))
                   ('interface  (format "public interface %s {\n\n}" class-name))
                   ('enum       (format "public enum %s {\n    ;\n}" class-name))
                   ('record     (format "public record %s() {\n}" class-name))
                   ('annotation (format "public @interface %s {\n\n}" class-name))
                   (_           (format "public class %s {\n\n}" class-name))))
           (content (if (string-empty-p package)
                        (concat body "\n")
                      (concat (format "package %s;\n\n%s\n" package body)))))
      (when (file-exists-p file)
        (user-error "File already exists: %s" file))
      (make-directory dir t)
      (find-file file)
      (insert content)
      (save-buffer)
      (goto-char (point-min))
      (search-forward "{" nil t)
      (forward-line 0)
      (open-line 1)
      (indent-for-tab-command)
      (message "Created %s %s" kind fqcn)))

  (defun my/java-new-class ()
    "Create a new Java class."
    (interactive)
    (let* ((pkg (my/java-current-package))
           (default (if (string-empty-p pkg) "" (concat pkg ".")))
           (fqcn (read-string "Class (e.g. com.foo.MyClass): " default)))
      (my/java-create-class fqcn 'class)))

           (default (if (string-empty-p pkg) "" (concat pkg ".")))

  (defun my/java-new-enum ()

           (default (if (string-empty-p pkg) "" (concat pkg ".")))
           (fqcn (read-string "Enum: " default)))

  (defun my/java-new-record ()

  (defun my/java-new-file ()
    "Create a new Java file, choosing the kind interactively (IntelliJ-style)."
    (interactive)
    (let* ((kind (intern (completing-read "Kind: " '("class" "interface" "enum" "record" "annotation") nil t)))
           (pkg (my/java-current-package))
           (default (if (string-empty-p pkg) "" (concat pkg ".")))
           (fqcn (read-string (format "New %s: " kind) default)))
      (my/java-create-class fqcn kind)))

  (defun my/java-go-to-test ()
    "Jump between source and test file (convention-based)."
    (interactive)
    (unless (buffer-file-name)
      (user-error "Not visiting a file"))
    (let* ((file (buffer-file-name))
           (root (project-root (project-current t)))
           (is-test (string-match-p "/src/test/" file)))
      (let ((target (if is-test
                        (replace-regexp-in-string "/src/test/" "/src/main/"
                          (replace-regexp-in-string "Test\\.java$" ".java" file))
                      (replace-regexp-in-string "/src/main/" "/src/test/"
                        (replace-regexp-in-string "\\.java$" "Test.java" file)))))
        (if (file-exists-p target)
            (find-file target)
          (when (y-or-n-p (format "Create %s?" (file-name-nondirectory target)))
            (make-directory (file-name-directory target) t)
            (find-file target)
            (let* ((rel (file-relative-name target (if is-test
                                                       (expand-file-name "src/main/java/" root)
                                                     (expand-file-name "src/test/java/" root))))
                   (parts (split-string (file-name-sans-extension rel) "/"))
                   (class-name (car (last parts)))
                   (pkg (mapconcat #'identity (butlast parts) ".")))
              (insert (format "package %s;\n\nimport org.junit.jupiter.api.Test;\n\nclass %s {\n\n    @Test\n    void shouldWork() {\n    }\n}\n" pkg class-name))
              (save-buffer)))))))

               (string-match-p "mise.*\\.toml\\'" (or buffer-file-name "")))

  (defun my/mise-edit-run-command ()
    "Edit the run command at point in an indirect buffer with proper LSP.
  Opens an indirect buffer with the embedded code, enabling full LSP features."
    (interactive)
    (let* ((line-content (buffer-substring-no-properties (line-beginning-position) (line-end-position)))
           (in-string (and (string-match "^run\\s-*=\\s-*\"\"\"\\(?:#!\\([^\"\n]+\\)\\)?" line-content)
                           (or (match-string 1 line-content) "/bin/bash"))))
      (goto-char (line-beginning-position))
      (if (re-search-forward "^run\\s-*=\\s-*\"\"\"\\(?:#!\\([^\"\n]+\\)\\)?" (line-end-position) t)
          (let* ((shebang (or (match-string 1) "/bin/bash"))
                 (lang (cond ((string-match-p "python" shebang) "python")
                             ((string-match-p "bash\\|sh" shebang) "bash")
                             ((string-match-p "node" shebang) "javascript")
                             (t "bash")))
                 (code-start (match-end 0))
                 (code-end (save-excursion
                             (goto-char code-start)
                             (if (re-search-forward "\"\"\"" nil t)
                                 (match-beginning 0)
                               (point-max))))
                 (code (buffer-substring-no-properties code-start code-end))
                 (buf-name (format "*mise-run:%s*" (file-name-nondirectory (buffer-file-name))))
                 (mode (intern (format "%s-ts-mode" lang))))
            (with-current-buffer (get-buffer-create buf-name)
              (erase-buffer)
              (insert code)
              (when (functionp mode) (funcall mode))
              (setq-local my/mise-source-buffer (current-buffer))
              (setq-local my/mise-source-start code-start)
              (setq-local my/mise-source-end code-end)
              (goto-char (point-min)))
            (pop-to-buffer buf-name)
            (message "Edit the command, then C-c C-c to save back to source"))
        (user-error "Not on a run command with multi-line string"))))

  (defun my/mise-save-back ()
    "Save edited content back to the source mise file."
    (interactive)
    (if (and (boundp 'my/mise-source-buffer)
             (buffer-live-p my/mise-source-buffer))
        (let ((content (buffer-string)))
          (with-current-buffer my/mise-source-buffer
            (goto-char my/mise-source-start)
            (delete-region my/mise-source-start my/mise-source-end)
            (insert content))
          (kill-buffer))
      (user-error "Not in a mise edit buffer")))

#+begin_src emacs-lisp
  (defvar my/java-coverage-overlays nil "Global list of active overlays.")

        (progn
          (my/java-coverage-clear)
          (setq-local my/java-coverage-visible nil)

      (my/java-apply-subtle-coverage)

  (defun my/java-apply-subtle-coverage ()
    "Apply lenses to the signature and orange highlights with tooltips in the body."
    (let ((report-path (my/java-get-jacoco-path))
          (root (project-root (project-current))))
      (when (and report-path root)
        (let* ((relative-path (file-relative-name (buffer-file-name) root))
               (xml (xml-parse-file report-path))
               (report (car xml)))
          (my/java-coverage-clear)

          (dolist (pkg (xml-get-children report 'package))
            ;; 1. Signature Lenses
            (dolist (cls (xml-get-children pkg 'class))
              (when (string-suffix-p (concat (xml-get-attribute cls 'name) ".java") relative-path)
                (dolist (method (xml-get-children cls 'method))
                  (let* ((line-num (string-to-number (xml-get-attribute method 'line)))
                         (counters (xml-get-children method 'counter))
                         (line-c (seq-find (lambda (c) (string= (xml-get-attribute c 'type) "LINE")) counters))
                         (m (string-to-number (or (xml-get-attribute line-c 'missed) "0")))
                         (c (string-to-number (or (xml-get-attribute line-c 'covered) "0")))
                         (pct (if (> (+ m c) 0) (/ (* c 100) (+ m c)) 0))
                         (icon (if (= pct 100) (nerd-icons-faicon "nf-fa-check_circle" :face 'success) ""))
                         (text (format "%s Coverage: %d%%" icon pct)))
                    (my/java--create-lens-at-signature line-num text)))))

            ;; 2. Orange Highlights on missed lines with Tooltip
            (dolist (src (xml-get-children pkg 'sourcefile))
              (when (string-suffix-p (xml-get-attribute src 'name) relative-path)
                (dolist (line (xml-get-children src 'line))
                  (let ((missed (string-to-number (xml-get-attribute line 'mi)))
                        (covered (string-to-number (xml-get-attribute line 'ci)))
                        (line-num (string-to-number (xml-get-attribute line 'nr))))

                      (my/java--highlight-missed-line
                       line-num
                       (format "JaCoCo: Missed %s instructions (Covered: %s)" missed covered))))))))))))

        (push ov my/java-coverage-overlays))))

          (push ov my/java-coverage-overlays)))))
