** Filtering (Orderless)
This enables "fuzzy" matching where space-separated terms match regardless of order.

#+begin_src emacs-lisp
(use-package orderless
  :ensure t
  :custom
  ;; Define estilos: Orderless como principal, Basic como fallback
  (completion-styles '(orderless basic))
  
  ;; Garante que arquivos usem "partial-completion" (ex: /u/b/ -> /usr/bin/)
  ;; O Orderless puro não é bom para expandir caminhos parciais
  (completion-category-defaults nil)
  (completion-category-overrides '((file (styles partial-completion))))
  
  :config
  ;; --- DISPATCHERS (O "Pulo do Gato") ---
  ;; Esta função permite usar sufixos/prefixos para alterar a busca dinamicamente:
  ;;  termo=  -> Busca exata (Literal)
  ;;  !termo  -> Exclui resultados com esse termo (Without)
  ;;  %termo  -> Ignora acentos/diacríticos (CharFold)
  ;;  ,termo  -> Inicialismo (ex: ,rmf acha read-multiple-files)
  
  (defun my/orderless-dispatch (pattern _index _total)
    (cond
     ;; Excluir termo se começar com !
     ((string-prefix-p "!" pattern) 
      `(orderless-without-literal . ,(substring pattern 1)))
     
     ;; Busca literal exata se terminar com =
     ((string-suffix-p "=" pattern) 
      `(orderless-literal . ,(substring pattern 0 -1)))
     
     ;; Inicialismo se começar com , (útil para comandos longos)
     ((string-prefix-p "," pattern) 
      `(orderless-initialism . ,(substring pattern 1)))
      
     ;; Flex (fuzzy "sujo") se terminar com ~
     ((string-suffix-p "~" pattern) 
      `(orderless-flex . ,(substring pattern 0 -1)))))

  ;; Aplica o dispatcher
  (setq orderless-style-dispatchers '(my/orderless-dispatch)))
  #+end_src
