#+AUTHOR: Nathan Martins Cunha
#+PROPERTY: header-args:emacs-lisp :tangle config.el :results silent

#+begin_src emacs-lisp
(setq custom-file (expand-file-name "custom.el" user-emacs-directory))
(when (file-exists-p custom-file)
  (load custom-file 'noerror))

(dolist (file '("core.el" "ui.el" "completion.el" "tools.el" "bindings.el"))
  (let ((path (expand-file-name file user-emacs-directory)))
    (when (file-exists-p path)
      (load path 'noerror 'nomessage))))
#+end_src
