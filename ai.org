*** AI Code Assistant (agent-shell)
#+begin_src emacs-lisp
  (use-package shell-maker
    :ensure t
    :demand t)

  (use-package acp
    :ensure t
    :demand t)

  (use-package agent-shell
    :ensure t
    :after (shell-maker acp)
    :config
    ;; ── Load OpenCode backend ─────────────────────────────────
    (require 'agent-shell-opencode)

    ;; Use OpenCode's own auth (already configured via `opencode auth login`)
    (setq agent-shell-opencode-authentication
          (agent-shell-opencode-make-authentication :none t))

    ;; Always use OpenCode, skip the selection prompt
    (setq agent-shell-preferred-agent-config 'opencode)

    ;; ── Terminal-like behavior ────────────────────────────────
    ;; Start fresh sessions without prompting (like opening a new terminal)
    (setq agent-shell-session-strategy 'new)

    ;; Instant interrupt without confirmation (like C-c in a raw TTY)
    (setq agent-shell-confirm-interrupt nil)

    ;; ── Visual chrome reduction ───────────────────────────────
    ;; Minimal text header instead of graphical blocks
    (setq agent-shell-header-style 'text)

    ;; Plain colored text labels instead of boxed UI elements
    (setq agent-shell-status-kind-label-function
          #'agent-shell--plain-colored-status-kind-label)

    ;; No busy spinner in mode line
    (setq agent-shell-show-busy-indicator nil)

    ;; ── Inline display (no collapsible sections) ──────────────
    ;; Auto-expand tool calls so you see everything inline
    (setq agent-shell-tool-use-expand-by-default t)

    ;; Auto-expand AI thought process sections
    (setq agent-shell-thought-process-expand-by-default t)

    ;; Auto-expand your own messages
    (setq agent-shell-user-message-expand-by-default t)

    ;; ── Keybindings ───────────────────────────────────────────
    :bind (:map agent-shell-mode-map
                ("C-l" . comint-clear-buffer)
                ("C-c C-k" . comint-clear-buffer)))

  ;; ── Agent Shell Sidebar ───────────────────────────────────
  (use-package agent-shell-sidebar
    :ensure nil
    :vc (:url "https://github.com/cmacrae/agent-shell-sidebar")
    :after agent-shell
    :config
    ;; Use OpenCode as the default provider in sidebar
    (setq agent-shell-sidebar-default-config
          (agent-shell-opencode-make-agent-config))

    ;; Sidebar on the right, 25% width, locked
    (setq agent-shell-sidebar-position 'right)
    (setq agent-shell-sidebar-width "25%")
    (setq agent-shell-sidebar-locked t))
 #+end_src
