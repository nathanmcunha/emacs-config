;; --- Doom-like "K" Documentation Lookup ---
(defun my/smart-lookup ()
  "Lookup documentation for the thing at point.
   1. Elisp: describe-symbol
   2. Org Src Block (Elisp): describe-symbol
   3. LSP / lsp-mode: lsp-describe-thing-at-point
   4. Fallback: evil-lookup"
  (interactive)
  (cond
   ;; 1. Emacs Lisp Mode
   ((derived-mode-p 'emacs-lisp-mode)
    (let ((sym (thing-at-point 'symbol)))
      (if sym (describe-symbol sym)
        (message "No symbol at point"))))

   ;; 2. Org Mode Source Blocks
   ((and (eq major-mode 'org-mode) (org-in-src-block-p))
    (let ((lang (car (org-babel-get-src-block-info)))
          (sym (thing-at-point 'symbol t)))
      (if (and (string= lang "emacs-lisp") sym)
          (describe-symbol (intern sym))
        (call-interactively #'evil-lookup))))

   ;; 3. LSP / lsp-mode
   ((bound-and-true-p lsp-mode)
    (lsp-describe-thing-at-point))

   ;; 4. Fallback
   (t
    (call-interactively #'evil-lookup))))

(with-eval-after-load 'evil
  (define-key evil-motion-state-map (kbd "K") #'my/smart-lookup))

(defun my/reload-config ()
  "Reload every tangled .el module after `config.org` is re-tangled.
Nix loads config.el which does `(dolist ... (load path))`; mirroring that loop
here keeps the runtime model identical so reload order, hooks, and advice stay consistent."
  (interactive)
  (dolist (file '("core.el" "ui.el" "theme.el" "completion.el" "tools.el"
                  "bindings.el" "notes.el" "containers.el" "ai.el" "eca.el"))
    (let ((path (expand-file-name file user-emacs-directory)))
      (when (file-exists-p path)
        (load path 'noerror 'nomessage))))
  (when (file-exists-p (expand-file-name "config.el" user-emacs-directory))
    (load (expand-file-name "config.el" user-emacs-directory) 'noerror 'nomessage))
  (message "Reloaded Emacs config."))

(defun my/move-this-file ()
  "Move or rename the current file to NEW-PATH."
  (interactive)
  (let* ((old-path (buffer-file-name))
         (new-path (read-file-name "Move file to: " old-path)))
    (if (not (and old-path (file-exists-p old-path)))
        (user-error "Buffer is not visiting any file")
      (let ((new-path (expand-file-name new-path)))
        (make-directory (file-name-directory new-path) t)
        (rename-file old-path new-path 1)
        (set-visited-file-name new-path t t)
        (message "File moved to %s" new-path)))))

(defun my/delete-this-file ()
  "Delete the current file and kill its buffer."
  (interactive)
  (let ((path (buffer-file-name)))
    (unless (and path (file-exists-p path))
      (user-error "Buffer is not visiting any file"))
    (when (y-or-n-p (format "Really delete %S?" (file-name-nondirectory path)))
      (delete-file path)
      (kill-buffer (current-buffer))
      (message "File deleted: %s" path))))

(defun my/create-file (filename)
  "Create a new file named FILENAME."
  (interactive "FCreate file: ")
  (if (file-exists-p filename)
      (find-file filename)
    (write-region "" nil filename)
    (find-file filename)
    (message "Created file: %s" filename)))

(defun my/copy-file-path ()
  "Copy the current file's path to the kill ring."
  (interactive)
  (let ((filename (or (buffer-file-name) (error "Buffer not visiting a file"))))
    (kill-new filename)
    (message "Copied: %s" filename)))

(defun my/copy-file-path-relative-to-project ()
  "Copy the current file's path relative to project root to the kill ring."
  (interactive)
  (let* ((filename (or (buffer-file-name) (error "Buffer not visiting a file")))
         (project-root (or (and (project-current) (project-root (project-current))) default-directory))
         (relative-path (file-relative-name filename project-root)))
    (kill-new relative-path)
    (message "Copied: %s" relative-path)))

(defun my/kill-all-buffers ()
  "Kill all buffers except the current one and special buffers
(`special-mode' derived buffers like help, magit, etc.)."
  (interactive)
  (let ((current-buffer (current-buffer)))
    (dolist (buffer (buffer-list))
      (unless (or (eq buffer current-buffer)
                  (with-current-buffer buffer
                    (derived-mode-p 'special-mode)))
        (kill-buffer buffer))))
  (message "Killed all buffers"))

(defun my/kill-other-buffers ()
  "Kill all buffers except the current one."
  (interactive)
  (let ((current-buffer (current-buffer)))
    (dolist (buffer (buffer-list))
      (unless (eq buffer current-buffer)
        (kill-buffer buffer))))
  (message "Killed other buffers"))

;; --- C++ Single-File Compile & Run (HackerRank Style) ---
(defun my/compile-and-run-cpp-file ()
  "Compile the current C++ file with C++23 and run the resulting binary."
  (interactive)
  (let* ((file (buffer-file-name))
         (out (file-name-sans-extension file)))
    (unless file
      (user-error "Buffer is not visiting a file"))
    (unless (or (derived-mode-p 'c++-mode)
                (derived-mode-p 'c++-ts-mode)
                (derived-mode-p 'c-mode)
                (derived-mode-p 'c-ts-mode))
      (user-error "Not in a C/C++ buffer"))
    (let ((cmd (format "g++ -std=c++23 -Wall -Wextra -O2 -o %s %s && %s"
                       (shell-quote-argument out)
                       (shell-quote-argument file)
                       (shell-quote-argument out))))
      (compile cmd))))

    "Bind KEY to COMMAND in `my-leader-map` with DESCRIPTION."
    (keymap-set my-leader-map key
                (if description
                    `(menu-item ,description ,command)
                  command)))

  (defun my/apply-evil-leader-bindings ()
    "Apply leader bindings for Evil and global fallback keys."
    (when (featurep 'evil)
