  (defun my/safe-set-font (families &rest props)
    "Safely cycle through FAMILIES and apply the first match with PROPS."
    (let ((found nil))
      (cl-loop for family in families
               when (member family (font-family-list))
               return (progn
                        (setq found t)
                        (set-face-attribute 'default nil :family family)
                        (cl-loop for (prop val) on props by 'cddr
                                 do (set-face-attribute 'default nil prop val))))
      (unless found
        (message "Warning: None of the specified font families are available: %s" families))))


  (defun my/setup-frame-visuals (frame)
    (with-selected-frame frame
      (scroll-bar-mode -1)
      (tool-bar-mode -1)
      (menu-bar-mode -1)

      ;; Apply fonts inside the frame creation hook to fix daemon loading
      (when (display-graphic-p frame)
        ;; Main coding font (Atkinson)
        (my/safe-set-font '("Atkinson Hyperlegible Mono" "FiraCode Nerd Font" "Hack") :height 110)

        ;; Proportional reading font (Lexend) for Org-mode
        (when (member "Lexend" (font-family-list))
          (set-face-attribute 'variable-pitch nil :family "Lexend" :height 1.15))

        ;; Symbol & Emoji fallbacks
        (dolist (charset '((symbol . "Symbols Nerd Font")
                           (emoji . "Noto Color Emoji")))
          (when (member (cdr charset) (font-family-list))
            (set-fontset-font t (car charset) (cdr charset)))))))

  (if (daemonp)
      (add-hook 'after-make-frame-functions #'my/setup-frame-visuals)
    (my/setup-frame-visuals (selected-frame)))

  ;; Rosé Pine — light & dark Doom variants
  ;; https://github.com/donniebreve/rose-pine-doom-emacs
  (defun my/rose-pine-apply-rainbow (variant)
    "Tune rainbow-identifiers for VARIANT (:light or :dark).
Rosé Pine Dawn is a warm light theme — identifiers need lower
saturation so they don't compete with the pastel background.
Rosé Pine (dark) uses higher lightness for legibility on the dark base."
    (if (eq variant :light)
        (setq rainbow-identifiers-cie-l-saturation 35
              rainbow-identifiers-cie-l-lightness 45)
      (setq rainbow-identifiers-cie-l-saturation 45
            rainbow-identifiers-cie-l-lightness 70)))

  (defun my/load-rose-pine (&optional variant)
    "Load Rosé Pine VARIANT (:light or :dark, default :light)."
    (let ((v (or variant :light)))
      (disable-theme 'doom-rose-pine)
      (disable-theme 'doom-rose-pine-dawn)
      (load-theme (if (eq v :light) 'doom-rose-pine-dawn 'doom-rose-pine) t)
      (my/rose-pine-apply-rainbow v)))

  ;; Default: light (Rosé Pine Dawn). Call (my/load-rose-pine :dark) to switch.
  (my/load-rose-pine :light)

    (defvar bionic-reading-face 'bold
      "Face used to highlight the fixation point of words.")

  (defun bionic-reading-apply-buffer ()
    "Apply the bionic fixation overlay to the entire buffer."
    (save-excursion
      (save-restriction
        (widen)
        (goto-char (point-min))
        (while (forward-word)
          (let* ((bounds (bounds-of-thing-at-point 'word))
                 (word-start (car bounds))
                 (word-end (cdr bounds))
                 (bold-len (1+ (/ (- word-end word-start) 2))))
            (put-text-property word-start (+ word-start bold-len) 
                               'font-lock-face bionic-reading-face))))))

  (defun bionic-reading-clear-buffer ()
  "Remove all bionic reading overlays from the buffer."
  (with-silent-modifications
    (remove-text-properties (point-min) (point-max) '(font-lock-face nil))))

;;;###autoload
(define-minor-mode bionic-reading-mode
  "Toggle Bionic Reading style overlays in the current buffer."
  :init-value nil
  :lighter " Bionic"
  (if bionic-reading-mode
      (bionic-reading-apply-buffer)
    (bionic-reading-clear-buffer)))
