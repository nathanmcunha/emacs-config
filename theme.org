  (defun my/set-font (family &rest props)
    "Set default face attributes if the family exists."
    (when (member family (font-family-list))
      (set-face-attribute 'default nil :family family :foundry "outline")))

  (defun my/safe-set-font (families &rest props)
    "Safely cycle through FAMILIES and apply the first match with PROPS."
    (cl-loop for family in families
             when (member family (font-family-list))
             return (progn
                      (set-face-attribute 'default nil :family family)
                      (cl-loop for (prop val) on props by 'cddr
                               do (set-face-attribute 'default nil prop val))))
    (unless (member (car families) (font-family-list))
      (message "Warning: None of the specified font families are available: %s" families)))


  (defun my/setup-frame-visuals (frame)
    (with-selected-frame frame
      (scroll-bar-mode -1)
      (tool-bar-mode -1)
      (menu-bar-mode -1)

      ;; Apply fonts inside the frame creation hook to fix daemon loading
      (when (display-graphic-p frame)
        ;; Main coding font (Atkinson)
        (my/safe-set-font '("Atkinson Hyperlegible Mono" "FiraCode Nerd Font" "Hack") :height 110)

        ;; Proportional reading font (Lexend) for Org-mode
        (when (member "Lexend" (font-family-list))
          (set-face-attribute 'variable-pitch nil :family "Lexend" :height 1.15))

        ;; Symbol & Emoji fallbacks
        (dolist (charset '((symbol . "Symbols Nerd Font")
                           (emoji . "Noto Color Emoji")))
          (when (member (cdr charset) (font-family-list))
            (set-fontset-font t (car charset) (cdr charset)))))))

  (if (daemonp)
      (add-hook 'after-make-frame-functions #'my/setup-frame-visuals)
    (my/setup-frame-visuals (selected-frame)))
