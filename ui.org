* Async Command Runner
A general-purpose async command runner with smart completion.
Detects mise tasks, npm scripts, Makefile targets, Gradle tasks, and Docker Compose commands
from the current project, and feeds them into the Corfu/Cape completion stack.
** Command Collectors
#+begin_src emacs-lisp
  (defvar my/run-history nil
    "History of shell commands run via my/run.")

  (defvar my/run-favorites
    '("mise run" "docker compose up -d" "docker compose down"
      "podman compose up -d" "podman compose down"
      "npm start" "npm test" "npm run build"
      "pytest" "python -m pytest"
      "cargo build" "cargo run" "cargo test"
      "make" "make test" "make clean")
    "Global favorite commands always shown in completion.")

  (defun my/run--mise-tasks ()
    "Return list of 'mise run TASK' strings from current project."
    (let ((default-directory (or (and (project-current)
                                      (project-root (project-current)))
                                 default-directory)))
      (condition-case nil
          (let ((output (shell-command-to-string "mise tasks --json 2>/dev/null")))
            (unless (string-empty-p output)
              (let* ((json-array-type 'list)
                     (json-object-type 'alist)
                     (parsed (json-read-from-string output)))
                (mapcar (lambda (task)
                          (let ((name (alist-get 'name task))
                                (desc (or (alist-get 'description task) "")))
                            (propertize (format "mise run %s" name)
                                       'my/annotation desc)))
                        parsed))))
        (error nil))))

  (defun my/run--package-json-scripts ()
    "Return list of 'npm run SCRIPT' strings from package.json."
    (let* ((root (or (and (project-current) (project-root (project-current)))
                     default-directory))
           (pkg (expand-file-name "package.json" root)))
      (when (file-exists-p pkg)
        (condition-case nil
            (let* ((json-object-type 'alist)
                   (json-array-type 'list)
                   (data (json-read-file pkg))
                   (scripts (alist-get 'scripts data)))
              (mapcar (lambda (s)
                        (propertize (format "npm run %s" (car s))
                                    'my/annotation (cdr s)))
                      scripts))
          (error nil)))))

  (defun my/run--makefile-targets ()
    "Return list of 'make TARGET' strings from Makefile."
    (let* ((root (or (and (project-current) (project-root (project-current)))
                     default-directory))
           (makefile (expand-file-name "Makefile" root)))
      (when (file-exists-p makefile)
        (condition-case nil
            (let ((output (shell-command-to-string
                           (format "make -pRrq -f %s : 2>/dev/null | awk -F: '/^[a-zA-Z0-9][^$#\\/\\t=]*:([^=]|$)/ {split($1,A,/ /);print A[1]}' | sort -u"
                                   (shell-quote-argument makefile)))))
              (mapcar (lambda (target) (format "make %s" target))
                      (split-string output "\n" t)))
          (error nil)))))

  (defun my/run--gradle-tasks ()
    "Return common Gradle tasks if gradlew exists."
    (let* ((root (or (and (project-current) (project-root (project-current)))
                     default-directory))
           (gradlew (expand-file-name "gradlew" root)))
      (when (file-exists-p gradlew)
        (mapcar (lambda (task) (format "./gradlew %s" task))
                '("build" "test" "run" "clean" "bootRun"
                  "assemble" "check" "classes" "jar")))))

  (defun my/run--docker-compose-services ()
    "Return docker compose commands if a compose file exists."
    (let* ((root (or (and (project-current) (project-root (project-current)))
                     default-directory))
           (compose-cmd (if (executable-find "podman") "podman compose" "docker compose")))
      (when (or (file-exists-p (expand-file-name "docker-compose.yml" root))
                (file-exists-p (expand-file-name "docker-compose.yaml" root))
                (file-exists-p (expand-file-name "compose.yml" root))
                (file-exists-p (expand-file-name "compose.yaml" root))
                (seq-some (lambda (f) (string-match-p "docker-compose" f))
                          (directory-files root nil "\\.ya?ml$")))
        (mapcar (lambda (sub) (format "%s %s" compose-cmd sub))
                '("up -d" "down" "logs -f" "ps" "build" "restart" "pull")))))

  (defvar my/run--cache nil
    "Alist of (PROJECT-ROOT TIMESTAMP CANDIDATES).")

  (defun my/run--project-key ()
    "Return project root or default-directory as key for caching."
    (or (and (project-current) (project-root (project-current))) default-directory))

  (defun my/run--project-markers-timestamp (root)
    "Return a stable timestamp (integer) computed from marker files in ROOT.
Checks common project files (package.json, Makefile, docker-compose.yml,
docker-compose.yaml, pom.xml, build.gradle), whose mtimes signal project
structure/config changes."
    (let* ((markers '("package.json" "Makefile" "docker-compose.yml" "docker-compose.yaml" "pom.xml" "build.gradle"))
           (times
            (cl-loop for f in markers
                     for path = (expand-file-name f root)
                     when (file-exists-p path)
                     collect (let* ((attrs (file-attributes path))
                                    (mtime (file-attribute-modification-time attrs)))
                               (truncate (float-time mtime))))))
      (if times (apply #'max times) 0)))

  (defun my/run--collect-candidates ()
    "Collect all command candidates for the current project with caching."
    (let* ((root (my/run--project-key))
           (stamp (when root (my/run--project-markers-timestamp root)))
           (entry (assoc root my/run--cache)))
      (if (and entry (equal (cadr entry) stamp))
          ;; cache hit: return cached candidates
          (caddr entry)
        ;; cache miss: compute and store
        (let ((cands (cl-remove-duplicates
                      (seq-filter #'stringp
                                  (append (my/run--mise-tasks)
                                          (my/run--package-json-scripts)
                                          (my/run--makefile-targets)
                                          (my/run--gradle-tasks)
                                          (my/run--docker-compose-services)
                                          my/run-favorites
                                          my/run-history))
                      :test #'string= :from-end t)))
          (setq my/run--cache (cl-delete root my/run--cache :key #'car :test #'equal))
          (push (list root stamp cands) my/run--cache)
          cands))))

  ;; Invalidate cache on common actions
  (advice-add 'project-switch-project :after (lambda (&rest _) (setq my/run--cache nil)))
  (advice-add 'save-buffer :after (lambda (&rest _) (setq my/run--cache nil)))

  (defun my/run--annotate (cand)
    "Annotation function for run candidates."
    (if-let ((ann (get-text-property 0 'my/annotation cand)))
        (format "  %s" (propertize ann 'face 'marginalia-documentation))
      ""))
#+end_src

** Runner Commands
#+begin_src emacs-lisp
  (defun my/run (cmd)
    "Run CMD asynchronously from project root, with smart completion."
    (interactive
     (let ((minibuffer-completing-file-name nil))
       (list (completing-read
              "Run: "
              (lambda (str pred action)
                (if (eq action 'metadata)
                    '(metadata (annotation-function . my/run--annotate)
                               (category . my/run-command))
                  (complete-with-action action (my/run--collect-candidates) str pred)))
              nil nil nil 'my/run-history))))
    (let* ((default-directory (or (and (project-current)
                                       (project-root (project-current)))
                                  default-directory))
           (short (car (split-string cmd)))
           (buf-name (format "*run:%s*" short)))
      (async-shell-command cmd buf-name)
      (message "Started: %s" cmd)))

  (defun my/run-compile (cmd)
    "Run CMD in a compilation buffer with smart completion."
    (interactive
     (list (completing-read
            "Compile: "
            (lambda (str pred action)
              (if (eq action 'metadata)
                  '(metadata (annotation-function . my/run--annotate)
                             (category . my/run-command))
                (complete-with-action action (my/run--collect-candidates) str pred)))
            nil nil nil 'my/run-history)))
    (let ((default-directory (or (and (project-current)
                                       (project-root (project-current)))
                                  default-directory)))
      (compile cmd)))

  (defun my/run-stop ()
    "Stop a running async command."
    (interactive)
    (let* ((bufs (seq-filter (lambda (b)
                               (and (string-prefix-p "*run:" (buffer-name b))
                                    (get-buffer-process b)))
                             (buffer-list)))
           (names (mapcar #'buffer-name bufs)))
      (if (null names)
          (message "No running commands")
        (let* ((choice (completing-read "Stop: " names nil t))
               (proc (get-buffer-process (get-buffer choice))))
          (when proc
            (interrupt-process proc)
            (sit-for 0.3)
            (when (process-live-p proc) (kill-process proc)))
          (message "Stopped %s" choice)))))

  (defun my/run-show ()
    "Switch to a running command buffer."
    (interactive)
    (let* ((bufs (seq-filter (lambda (b)
                               (string-prefix-p "*run:" (buffer-name b)))
                             (buffer-list)))
           (names (mapcar #'buffer-name bufs)))
      (if (null names)
          (message "No command buffers")
        (switch-to-buffer (completing-read "Show: " names nil t)))))

  (defun my/run-restart ()
    "Restart the last command from history."
    (interactive)
    (if my/run-history
        (my/run (car my/run-history))
      (call-interactively #'my/run)))

** Cape Integration (In-Buffer Completion)
Provides Corfu popup suggestions for shell commands in eshell, shell-mode,
comint, eat, vterm, and agent-shell buffers.

#+begin_src emacs-lisp
  (defvar my/cape-run-cache nil
    "Cached candidates as (PROJECT-ROOT . CANDIDATES).")

  (defun my/cape-run--invalidate-cache ()
    "Clear cache when switching projects."
    (setq my/cape-run-cache nil))

  (advice-add 'project-switch-project :after
              (lambda (&rest _) (my/cape-run--invalidate-cache)))

  (defun my/cape-run--candidates ()
    "Return cached or fresh command candidates for the current project."
    (let ((root (or (and (project-current) (project-root (project-current)))
                    default-directory)))
      (if (and my/cape-run-cache (string= (car my/cape-run-cache) root))
          (cdr my/cape-run-cache)
        (let ((cands (my/run--collect-candidates)))
          (setq my/cape-run-cache (cons root cands))
          cands))))

  (defun my/cape-run-capf ()
    "Completion-at-point function for shell commands.
  Completes from BOL or after &&, ||, |, ; separators."
    (let* ((line-start (line-beginning-position))
           (pt (point))
           (text (buffer-substring-no-properties line-start pt))
           ;; Find where the current command starts (after pipe, &&, ;, etc.)
           (cmd-start (if (string-match ".*\\(?:[;&|][ \t]*\\)" text)
                          (+ line-start (match-end 0))
                        line-start))
           (prefix (buffer-substring-no-properties cmd-start pt)))
      (when (>= (length prefix) 1)
        (list cmd-start pt
              (completion-table-dynamic
               (lambda (_) (my/cape-run--candidates)))
              :annotation-function #'my/run--annotate
              :exclusive 'no))))

  ;; Hook into shell/terminal modes (low priority so LSP/native CAPFs win)
  (dolist (hook '(eshell-mode-hook
                  shell-mode-hook
                  comint-mode-hook
                  eat-mode-hook
                  vterm-mode-hook
                  agent-shell-mode-hook))
    (add-hook hook
              (lambda ()
                (add-hook 'completion-at-point-functions
                          #'my/cape-run-capf -50 t))))

  ;; Also provide completion in M-x shell-command, compile, and my/run prompts
  (add-hook 'minibuffer-setup-hook
            (lambda ()
              (when (memq this-command
                          '(async-shell-command shell-command compile
                            my/run my/run-compile))
                (add-hook 'completion-at-point-functions
                          #'my/cape-run-capf -50 t))))

  ;; Merged shell super-CAPF: project commands + files + history + dabbrev
  ;; This replaces the default CAPFs in shell buffers with a combined source.
  (defun my/cape-shell-super-capf ()
    "Combined CAPF: project commands + file paths + history + dabbrev."
    (cape-wrap-super
     #'my/cape-run-capf
     #'cape-file
     #'cape-history
     #'cape-dabbrev))

  (dolist (hook '(eshell-mode-hook
                  shell-mode-hook
                  comint-mode-hook))
    (add-hook hook
              (lambda ()
                (setq-local completion-at-point-functions
                            (list #'my/cape-shell-super-capf)))))

** Popup Windows (built-in toggle)
#+begin_src emacs-lisp
  (defvar my/popup-buffer-regexps
    '("\\*Messages\\*" "Output\\*$" "\\*Async Shell Command\\*"
      "\\*Help\\*" "\\*compilation\\*" "\\*Flymake diagnostics\\*" "\\*Warnings\\*"
      "\\*docker-compose\\*" "\\*docker-run\\*")
    "Regexps matching buffers treated as popups.")

  (defun my/popup-buffer-p (buf)
    "Return t if BUF matches a popup regexp or is derived from a popup mode."
    (let ((name (buffer-name buf)))
      (or (seq-some (lambda (re) (string-match-p re name)) my/popup-buffer-regexps)
          (with-current-buffer buf
            (derived-mode-p 'help-mode 'compilation-mode)))))

  (defun my/popup-toggle ()
    "Hide any visible popup, or show the most recent one."
    (interactive)
    (let ((popup-win (seq-find (lambda (w) (my/popup-buffer-p (window-buffer w)))
                               (window-list))))
      (if popup-win
          (delete-window popup-win)
        (when-let ((buf (seq-find #'my/popup-buffer-p (buffer-list))))
          (display-buffer buf)))))

  (defun my/popup-cycle ()
    "Cycle through popup buffers."
    (interactive)
    (let* ((popups (seq-filter #'my/popup-buffer-p (buffer-list)))
           (current (and (my/popup-buffer-p (current-buffer)) (current-buffer)))
           (next (if current
                     (or (cadr (member current popups)) (car popups))
                   (car popups))))
      (when next (display-buffer next))))

  (global-set-key (kbd "C-`")   #'my/popup-toggle)
  (global-set-key (kbd "M-`")   #'my/popup-cycle)
#+end_src
