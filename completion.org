    :ensure t
    :custom
    (completion-styles '(orderless basic))
    (completion-category-overrides '((file (styles basic partial-completion))))
    :config
    (defun my/orderless-dispatch (pattern _index _total)
      (cond
       ;; Standard dispatchers
       ((string-prefix-p "!" pattern) `(orderless-without-literal . ,(substring pattern 1)))
       ((string-suffix-p "=" pattern) `(orderless-literal . ,(substring pattern 0 -1)))
       ((string-prefix-p "," pattern) `(orderless-initialism . ,(substring pattern 1)))
       ((string-suffix-p "~" pattern) `(orderless-flex . ,(substring pattern 0 -1)))
       ;; New: Anchor dispatchers
       ((string-prefix-p "^" pattern) `(orderless-regexp . ,(concat "^" (substring pattern 1))))
       ((string-suffix-p "$" pattern) `(orderless-regexp . ,(concat (substring pattern 0 -1) "$")))))
    (setq orderless-style-dispatchers '(my/orderless-dispatch)))

  (use-package marginalia
    :ensure t
    :after vertico
    :custom
    (marginalia-max-relative-age 0)
    (marginalia-align 'right)
    :init (marginalia-mode)
    :bind (:map minibuffer-local-map ("M-A" . marginalia-cycle))
    :config
    ;; Custom simplified annotator for Buffers
    (defun my/marginalia-annotate-buffer (cand)
      (when-let (buffer (get-buffer cand))
        (let* ((file (buffer-file-name buffer))
               (type (cond (file "File")
                           ((derived-mode-p 'dired-mode) "Dir")
                           (t "Buf")))
               ;; Get path relative to project or home
               (path (if file
                         (if-let ((project (project-current nil (file-name-directory file))))
                             (file-relative-name file (project-root project))
                           (abbreviate-file-name file))
                       "")))
          (marginalia--fields
           (type :width 5 :face 'marginalia-type)
           (path :face 'marginalia-file-name)))))

    ;; Custom simplified annotator for Files (Recentf)
    (defun my/marginalia-annotate-file (cand)
      (let* ((path (if-let ((project (project-current nil (file-name-directory cand))))
                       (file-relative-name cand (project-root project))
                     (abbreviate-file-name cand))))
        (marginalia--fields
         ("File" :width 5 :face 'marginalia-type)
         (path :face 'marginalia-file-name))))

    ;; Register the custom annotators safely
    (with-eval-after-load 'marginalia
      (setq marginalia-annotator-alist
            (append '((buffer . my/marginalia-annotate-buffer)
                      (file . my/marginalia-annotate-file))
                    (bound-and-true-p marginalia-annotator-alist)))))
