* Completion System (Vertico Stack)
Based on Doom Emacs configuration patterns for a clean, IDE-like experience.

** Vertico (Minibuffer Completion)
#+begin_src emacs-lisp
  (use-package vertico
    :demand t
    :config (vertico-mode)
    :custom
    (vertico-cycle t)

                ("C-j" . vertico-next)

                ([tab] . minibuffer-complete)
                ("C-l" . vertico-exit)
                ("RET" . vertico-exit)
                ("M-RET" . vertico-exit-input)
                ("C-f" . vertico-scroll-down)

    :config

    (require 'vertico-directory)

    (define-key vertico-map (kbd "DEL") #'vertico-directory-delete-char)
    (define-key vertico-map (kbd "M-DEL") #'vertico-directory-delete-word)

** Vertico Multiform (Visual Enhancements)
#+begin_src emacs-lisp
  ;; Doom-like: highlight directories and enabled modes in completion
  (use-package vertico-multiform
    :hook (vertico-mode . vertico-multiform-mode)
    :config
    (defun +vertico-highlight-directory (file)
      "Highlight directories with the marginalia directory face."
      (when (string-suffix-p "/" file)
        (add-face-text-property 0 (length file) 'marginalia-file-priv-dir 'append file))
      file)

    (defun +vertico-highlight-enabled-mode (cmd)
      "Highlight enabled major/minor modes in M-x."
      (let ((sym (intern cmd)))
        (with-current-buffer (nth 1 (buffer-list))
          (when (or (eq sym major-mode)
                    (and (memq sym minor-mode-list)
                         (boundp sym)
                         (symbol-value sym)))
            (add-face-text-property 0 (length cmd) 'font-lock-constant-face 'append cmd)))
        cmd))

    (add-to-list 'vertico-multiform-categories
                 '(file (+vertico-transform-functions . +vertico-highlight-directory)))
    (add-to-list 'vertico-multiform-commands
                 '(execute-extended-command (+vertico-transform-functions . +vertico-highlight-enabled-mode))))
#+end_src

* In-Buffer Completion (Corfu + Cape)
Based on Doom Emacs corfu configuration for VS Code-like IntelliSense.

** Corfu (IntelliSense Popup)
#+begin_src emacs-lisp
  (use-package corfu
    :demand t
    :custom
    (corfu-auto t)
    (corfu-auto-delay 0.5)            ; pop only when user pauses
    (corfu-auto-prefix 3)             ; wait for 3 chars
    (corfu-cycle t)
    (corfu-preselect 'prompt)
    (corfu-count 16)
    (corfu-max-width 120)
    (corfu-on-exact-match nil)
    (corfu-quit-at-boundary 'separator)
    (corfu-quit-no-match 'separator)
    (corfu-popupinfo-delay '(0.8 . 0.4))
    :init
    (setq tab-always-indent 'complete)
    :bind
    (:map corfu-map
          ("C-j" . corfu-next)
          ("C-k" . corfu-previous)
          ("C-l" . corfu-insert)
          ("C-h" . corfu-reset)
          ("TAB" . corfu-next)
          ([tab] . corfu-next)
          ("S-TAB" . corfu-previous)
          ([backtab] . corfu-previous)
          ("RET" . corfu-insert)
          ("M-d" . corfu-popupinfo-toggle)
          ("M-p" . corfu-popupinfo-scroll-up)
          ("M-n" . corfu-popupinfo-scroll-down))
    :config
    (global-corfu-mode)
    (keymap-global-set "C-." #'completion-at-point)
    (keymap-global-set "C-SPC" #'completion-at-point)

    ;; Use orderless for LSP completions (like Doom)
    (add-to-list 'completion-category-overrides `(lsp-capf (styles orderless)))

    ;; Exclude modes where Corfu shouldn't activate
    (setq global-corfu-modes
          '((not erc-mode circe-mode help-mode gud-mode ghostel-mode magit-mode)
            t)))

  (use-package corfu-popupinfo
    :hook ((corfu-mode . corfu-popupinfo-mode)))

  (use-package corfu-history
    :hook ((corfu-mode . corfu-history-mode))
    :config
    (with-eval-after-load 'savehist
      (add-to-list 'savehist-additional-variables 'corfu-history)))

  ;; Nerd Icons for Corfu
  (use-package nerd-icons-corfu
    :after corfu
    :config
    (add-to-list 'corfu-margin-formatters #'nerd-icons-corfu-formatter))

** Cape (Completion Backends)
#+begin_src emacs-lisp
  (use-package cape
    :demand t
    :init
    ;; Make LSP completion non-interruptible (key fix for stability)
    (advice-add #'lsp-completion-at-point :around #'cape-wrap-noninterruptible)
    (advice-add #'comint-completion-at-point :around #'cape-wrap-nonexclusive)
    (advice-add #'eglot-completion-at-point :around #'cape-wrap-nonexclusive)
    (advice-add #'pcomplete-completions-at-point :around #'cape-wrap-nonexclusive)

    ;; File completion in prog modes (LOW priority = runs after LSP)
    (add-hook 'prog-mode-hook
              (lambda () (add-hook 'completion-at-point-functions #'cape-file -10 t)))

    ;; Elisp in org/markdown
    (add-hook 'org-mode-hook
              (lambda () (add-hook 'completion-at-point-functions #'cape-elisp-block 0 t)))
    (add-hook 'markdown-mode-hook
              (lambda () (add-hook 'completion-at-point-functions #'cape-elisp-block 0 t)))


    ;; Dabbrev as FALLBACK (high priority = runs last)
    (dolist (hook '(prog-mode-hook text-mode-hook conf-mode-hook comint-mode-hook))
      (add-hook hook
                (lambda () (add-hook 'completion-at-point-functions #'cape-dabbrev 20 t))))

    (setq completion-ignore-case t)
    (setq cape-dabbrev-check-other-buffers nil)  ; current buffer only
    (setq cape-dabbrev-buffer-scanning-size-limit (* 256 1024)))  ; 256KB limit
#+end_src
