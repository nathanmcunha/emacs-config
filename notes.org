        (defun my/org-config-file-p (&optional file)
          "Return non-nil when FILE is one of the managed literate config files."
          (let* ((f (or file buffer-file-name))
                 (name (and f (file-name-nondirectory f)))
                 (in-config-dir (and f
                                     (string-prefix-p (file-truename user-emacs-directory)
                                                      (file-truename (file-name-directory f))))))
            (and in-config-dir (member name my/org-auto-tangle-files))))

        (defun my/org-auto-tangle-config ()
          "Auto-tangle the current Org buffer when it belongs to Emacs config."
          (when (my/org-config-file-p)
            (let ((org-confirm-babel-evaluate nil))
              (org-babel-tangle)
              (message "Auto-tangled %s" (file-name-nondirectory buffer-file-name)))))

        (defun my/org-enable-auto-tangle ()
          "Enable per-buffer auto-tangle for managed Emacs config Org files."
          (when (my/org-config-file-p)
            (add-hook 'after-save-hook #'my/org-auto-tangle-config nil t)))
